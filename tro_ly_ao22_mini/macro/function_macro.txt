FUNCTION excel_create(name)
    excel -f $name.xlsx create
ENDFUNCTION

FUNCTION excel_create2(name)
    PYTHON __import__("os").path.exists("{name}.xlsx") -> exists
    ASSERT {exists} == FALSE, "File đã tồn tại"
    excel -f {name}.xlsx create
ENDFUNCTION
