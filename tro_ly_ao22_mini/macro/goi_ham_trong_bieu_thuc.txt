FUNCTION add(a, b)
    RETURN a + b
ENDFUNCTION

FUNCTION mul(a, b=2)
    RETURN a * b
ENDFUNCTION

FUNCTION square(x)
    RETURN mul(x, x)
ENDFUNCTION

FUNCTION combo(a, b)
    RETURN add(square(a), mul(b, 3))
ENDFUNCTION
