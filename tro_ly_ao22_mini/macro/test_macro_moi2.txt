PRINT "=== Train XOR Neural Network ==="

# ==========================================
# Dataset XOR
# ==========================================

SET X = [[0,0], [0,1], [1,0], [1,1]]
SET y = [0,1,1,0]

# ==========================================
# Hyperparameters
# ==========================================

SET lr = 0.8
SET epochs = 80

# ==========================================
# Initialize weights
# ==========================================

# Hidden neuron h1
SET w1 = 0.1
SET w2 = 0.2
SET b1 = 0.3

# Hidden neuron h2
SET w3 = 0.4
SET w4 = 0.5
SET b2 = 0.6

# Output neuron
SET w5 = 0.7
SET w6 = 0.8
SET b3 = 0.9

# ==========================================
# Training loop
# ==========================================

LOOP $epochs

    SET total_loss = 0

    FOREACH x, target IN zip($X, $y)

        # ==================================
        # INPUT
        # ==================================

        SET x1 = ${x[0]}
        SET x2 = ${x[1]}

        # ==================================
        # FORWARD PASS
        # ==================================

        # Hidden neuron h1
        SET z1 = $w1*$x1 + $w2*$x2 + $b1
        SET h1 = 1 / (1 + math.exp(-$z1))

        # Hidden neuron h2
        SET z2 = $w3*$x1 + $w4*$x2 + $b2
        SET h2 = 1 / (1 + math.exp(-$z2))

        # Output neuron
        SET z3 = $w5*$h1 + $w6*$h2 + $b3
        SET pred = 1 / (1 + math.exp(-$z3))

        # ==================================
        # LOSS
        # ==================================

        SET loss = ($target - $pred) ** 2
        SET total_loss = $total_loss + $loss

        # ==================================
        # BACKPROP
        # ==================================

        # dLoss/dPred
        SET d_pred = -2 * ($target - $pred)

        # sigmoid derivative output
        SET sig3 = $pred * (1 - $pred)

        # output delta
        SET delta3 = $d_pred * $sig3

        # gradients output layer
        SET grad_w5 = $delta3 * $h1
        SET grad_w6 = $delta3 * $h2
        SET grad_b3 = $delta3

        # ==================================
        # Hidden layer gradients
        # ==================================

        # hidden neuron h1
        SET sig1 = $h1 * (1 - $h1)
        SET delta1 = $delta3 * $w5 * $sig1

        # hidden neuron h2
        SET sig2 = $h2 * (1 - $h2)
        SET delta2 = $delta3 * $w6 * $sig2

        # gradients h1
        SET grad_w1 = $delta1 * $x1
        SET grad_w2 = $delta1 * $x2
        SET grad_b1 = $delta1

        # gradients h2
        SET grad_w3 = $delta2 * $x1
        SET grad_w4 = $delta2 * $x2
        SET grad_b2 = $delta2

        # ==================================
        # UPDATE WEIGHTS
        # ==================================

        SET w1 = $w1 - $lr * $grad_w1
        SET w2 = $w2 - $lr * $grad_w2
        SET b1 = $b1 - $lr * $grad_b1

        SET w3 = $w3 - $lr * $grad_w3
        SET w4 = $w4 - $lr * $grad_w4
        SET b2 = $b2 - $lr * $grad_b2

        SET w5 = $w5 - $lr * $grad_w5
        SET w6 = $w6 - $lr * $grad_w6
        SET b3 = $b3 - $lr * $grad_b3

    ENDFOREACH

    # ==================================
    # LOG
    # ==================================

    IF ${loop_index} % 1 == 0
        PRINT "Epoch ${loop_index}: loss = ${total_loss}"
    ENDIF

ENDLOOP

# ==========================================
# PRINT FINAL WEIGHTS
