FUNCTION demo_pack(a, b, c)
    PRINT "a={a}, b={b}, c={c}"
ENDFUNCTION

SET arr = [1, 2, 3]

SET data = [(1,2), (3,4), (5,6)]

FOREACH a, b IN data
    PRINT "{a} + {b} = {a + b}"
ENDFOREACH
CALL demo_pack(*arr)
