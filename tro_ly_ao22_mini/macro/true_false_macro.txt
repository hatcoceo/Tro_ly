SET value = 42
IF value > 10 AND TRUE
    PRINT "✅ TRUE được dùng, điều kiện đúng"
ELSE
    PRINT "❌ Không chạy"
ENDIF
IF FALSE
    PRINT "❌ FALSE làm điều kiện sai"
ELSE
    PRINT "✅ ELSE chạy vì điều kiện FALSE"
ENDIF
IF value < 10 OR FALSE
    PRINT "❌ Cả hai đều FALSE nên không vào IF"
ELSE
    PRINT "✅ Điều kiện sai, rẽ vào ELSE"
ENDIF
IF NOT FALSE
    PRINT "✅ NOT FALSE tương đương TRUE"
ENDIF
IF TRUE AND (value == 42)
    PRINT "✅ Kết hợp TRUE với biểu thức đúng"
ENDIF
